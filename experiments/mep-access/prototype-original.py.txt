def sha(path):return hashlib.sha256(pathlib.Path(path).read_bytes()).hexdigest()
def write(path,x):pathlib.Path(path).write_text(json.dumps(x,indent=2))
def wrap(a):return (a+math.pi)%(2*math.pi)-math.pi

def scenarios():
 out=[]

def transform(q,c):
 a=c['angle'];co=math.cos(a);si=math.sin(a)
 return [co*q[0]-si*q[1]+c['offset'][0],si*q[0]+co*q[1]+c['offset'][1],wrap(q[2]+a)]

  qs.extend([[c['goal'][0],c['goal'][1],0],c['goal']]);metadata['manual_routes'].append([transform(q,c) for q in qs])

class Scene:
 def __init__(self,name):
  d=ROOT/'fixtures';self.meta=json.loads((d/(name+'.json')).read_text());self.name=name;self.f=ifcopenshell.open(d/(name+'.ifc'))
  self.unit_scale=ifcopenshell.util.unit.calculate_unit_scale(self.f)
  if self.unit_scale not in (1.,.001):raise ValueError('unsupported unit')
  settings=ifcopenshell.geom.settings();settings.set('use-world-coords',True);settings.set('convert-back-units',False)
  self.shapes={}
  for el in self.f.by_type('IfcBuildingElementProxy'):
   shape=ifcopenshell.geom.create_shape(settings,el);v=np.asarray(shape.geometry.verts).reshape(-1,3)
   if len(v)<4 or not np.isfinite(v).all():raise ValueError('missing/nonfinite geometry')
   self.shapes[el.GlobalId]={'polygon':MultiPoint(v[:,:2]).convex_hull,'zmin':float(v[:,2].min()),'zmax':float(v[:,2].max())}
  expected=set(self.meta['ids'].values())
  if set(self.shapes)!=expected:raise ValueError('missing/unknown solid')
  self.floor=self.shapes[self.meta['ids']['floor']]['polygon'];equipment=self.shapes[self.meta['ids']['equipment']]
  self.start=self.meta['start'];self.goal=self.meta['goal']
  local=rotate(translate(equipment['polygon'],-self.start[0],-self.start[1]),-self.start[2],origin=(0,0),use_radians=True)
  self.body=local;bounds=local.bounds;self.radius=max(math.hypot(x,y) for x,y in local.exterior.coords)
  self.height=equipment['zmax']-equipment['zmin'];self.bottom=equipment['zmin'];self.top=equipment['zmax']
  self.obstacles=[];self.optional=[]
  for gid,info in self.meta['interventions'].items():
   sh=self.shapes[gid];active=not(sh['zmin']>self.top+CLEAR+TOL or sh['zmax']<self.bottom-CLEAR-TOL)
   o={**info,**sh,'gid':gid,'active':active}
   if info['allowed']:
    if info['cost'] is None or info['cost']<=0:raise ValueError('unknown/nonpositive cost')
    o['bit']=1<<len(self.optional);self.optional.append(o)
   else:o['bit']=0
   self.obstacles.append(o)
  self.hard=[o for o in self.obstacles if o['active'] and not o['allowed']]
  self.calls=0;self.uncertain=0
 def pose(self,q):return translate(rotate(self.body,q[2],origin=(0,0),use_radians=True),q[0],q[1])
 def state_valid(self,q):
  body=self.pose(q)
  return self.floor.covers(body) and body.distance(self.floor.boundary)>CLEAR+TOL and all(body.distance(o['polygon'])>CLEAR+TOL for o in self.hard)
 def sweep_separate(self,a,b,obstacle=None):
  """Conservative Lipschitz displacement bound, independently of OMPL validation."""
  def rec(a,b,depth):
   self.calls+=1;da=wrap(b[2]-a[2]);mid=[(a[0]+b[0])/2,(a[1]+b[1])/2,wrap(a[2]+da/2)]
   body=self.pose(mid);B=math.hypot(b[0]-a[0],b[1]-a[1])/2+2*self.radius*math.sin(abs(da)/4)
   if obstacle is None:
    if not self.floor.covers(body):return False
    dist=body.distance(self.floor.boundary)
   else:dist=body.distance(obstacle)
   if dist<=CLEAR+TOL:return False
   if dist>B+CLEAR+TOL:return True
   if B<=.001 or depth>=20:self.uncertain+=1;return False
   return rec(a,mid,depth+1) and rec(mid,b,depth+1)
  return rec(a,b,0)
 def labels(self,a,b):
  if not self.sweep_separate(a,b):return None
  mask=0
  for o in self.obstacles:
   if not o['active']:continue
   if not self.sweep_separate(a,b,o['polygon']):
    if not o['allowed']:return None
    mask|=o['bit']
  return mask
 def cost(self,mask):return sum(o['cost'] for o in self.optional if mask&o['bit'])
 def names(self,mask):return [o['name'] for o in self.optional if mask&o['bit']]
 def length(self,a,b):return math.hypot(b[0]-a[0],b[1]-a[1])+self.radius*abs(wrap(b[2]-a[2]))
 def verify_path(self,qs,mask):
  for a,b in zip(qs,qs[1:]):
   labels=self.labels(a,b)
   if labels is None or labels&~mask:return False
  return True

def roadmap(scene,seed):
 ou.RNG.setSeed(seed);space=ob.SE2StateSpace();bounds=ob.RealVectorBounds(2);bb=scene.floor.bounds
 for i in range(2):bounds.setLow(i,bb[i]);bounds.setHigh(i,bb[i+2])
 space.setBounds(bounds);si=ob.SpaceInformation(space)
 def valid(s):return scene.state_valid([s.getX(),s.getY(),s.getYaw()])
 si.setStateValidityChecker(valid);si.setStateValidityCheckingResolution(.005);si.setup();states=[]
 for q in [scene.start,scene.goal]:
  s=space.allocState();s.setXY(q[0],q[1]);s.setYaw(q[2]);states.append(s)
 problem=ob.ProblemDefinition(si);problem.setStartAndGoalStates(*states);planner=og.PRM(si);planner.setProblemDefinition(problem);planner.setup()
 begin=time.monotonic();status=planner.solve(ob.timedPlannerTerminationCondition(0.0));planner.constructRoadmap(ob.timedPlannerTerminationCondition(2.0))
 pdata=ob.PlannerData(si);planner.getPlannerData(pdata);nodes=[];raw=[];starts=[];goals=[]
 for i in range(pdata.numVertices()):
  s=pdata.getVertex(i).getState();nodes.append([s.getX(),s.getY(),s.getYaw()])
  if pdata.isStartVertex(i):starts.append(i)
  if pdata.isGoalVertex(i):goals.append(i)
  for j in pdata.getEdges(i):
   if i<j:raw.append([i,j])
 return {'nodes':nodes,'raw_edges':raw,'starts':starts,'goals':goals,'planner_seconds':time.monotonic()-begin,'query_initial_status':str(status),'seed':seed,'planner':'OMPL PRM2.0.1','case':scene.name}

def label_graph(scene,g):
 edges=[];rejected=0;t=time.monotonic()
 for i,j in g['raw_edges']:
  mask=scene.labels(g['nodes'][i],g['nodes'][j])
  if mask is None:rejected+=1;continue
  edges.append([i,j,mask,scene.length(g['nodes'][i],g['nodes'][j])])
 g.update(edges=edges,rejected_edges=rejected,checking_seconds=time.monotonic()-t,checker_calls=scene.calls,uncertain_intervals=scene.uncertain)
 return g

def adjacency(g):
 adj=[[] for _ in g['nodes']]
 for i,j,m,l in g['edges']:adj[i].append((j,m,l));adj[j].append((i,m,l))
 return adj

def path_search(g,adj,allowed=None,greedy=None):
 targets=set(g['goals']);heap=[];dist={};prev={}
 for s in g['starts']:dist[s]=(0.,0.);heapq.heappush(heap,(0.,0.,s))
 while heap:
  a,b,v=heapq.heappop(heap)
  if (a,b)!=dist[v]:continue
  if v in targets:
   path=[v];mask=0
   while v in prev:v,m=prev[v];mask|=m;path.append(v)
   return {'path':list(reversed(path)),'mask':mask,'length':b,'objective':a}
  for w,m,length in adj[v]:
   if allowed is not None and m&~allowed:continue
   primary=greedy(m) if greedy else 0;new=(a+primary,b+length)
   if new<dist.get(w,(math.inf,math.inf)):dist[w]=new;prev[w]=(v,m);heapq.heappush(heap,(*new,w))
 return None

def joint(scene,g,adj):
 start=time.monotonic();targets=set(g['goals']);heap=[];dist={};prev={};expanded=0
 for s in g['starts']:dist[(s,0)]=(0.,0.);heapq.heappush(heap,(0.,0.,s,0))
 while heap:
  cost,length,v,mask=heapq.heappop(heap)
  if (cost,length)!=dist[(v,mask)]:continue
  expanded+=1
  if expanded>100000 or time.monotonic()-start>60:return {'status':'censored','expanded':expanded}
  if v in targets:
   state=(v,mask);path=[v]
   while state in prev:state=prev[state];path.append(state[0])
   return {'status':'found','cost':cost,'length':length,'mask':mask,'path':list(reversed(path)),'expanded':expanded,'seconds':time.monotonic()-start}
  for w,labels,l in adj[v]:
   newmask=mask|labels;new=(scene.cost(newmask),length+l);state=(w,newmask)
   if new<dist.get(state,(math.inf,math.inf)):dist[state]=new;prev[state]=(v,mask);heapq.heappush(heap,(*new,w,newmask))
 return {'status':'no_roadmap_route','expanded':expanded,'seconds':time.monotonic()-start}

def comparisons(scene,g):
 adj=adjacency(g);optimal=joint(scene,g,adj);oracle=None;oracle_rows=[]
 for mask in range(1<<len(scene.optional)):
  answer=path_search(g,adj,allowed=mask);cost=scene.cost(mask)
  oracle_rows.append({'mask':mask,'cost':cost,'found':answer is not None})
  if answer and (oracle is None or (cost,answer['length'])<(oracle['cost'],oracle['length'])):oracle={**answer,'allowed_mask':mask,'cost':cost}
 greedy=[]
 for mode in ['count','cost']:
  removed=0;steps=[]
  for _ in range(len(scene.optional)+1):
   metric=(lambda m:(m&~removed).bit_count()) if mode=='count' else (lambda m:scene.cost(m&~removed))
   route=path_search(g,adj,greedy=metric)
   if route is None:break
   pending=route['mask']&~removed
   if not pending:break
   choice=min((o for o in scene.optional if pending&o['bit']),key=lambda o:(o['cost'],o['gid']));removed|=choice['bit'];steps.append(choice['name'])
  greedy.append({'mode':mode,'found':route is not None and not(route['mask']&~removed),'cost':scene.cost(removed),'mask':removed,'steps':steps,'path':route['path'] if route else None})
 manual=[]
 for index,qs in enumerate(scene.meta['manual_routes']):
  mask=0;valid=True
  for a,b in zip(qs,qs[1:]):
   labels=scene.labels(a,b)
   if labels is None:valid=False;break
   mask|=labels
  manual.append({'index':index,'valid':valid,'cost':scene.cost(mask) if valid else None,'mask':mask if valid else None,'states':qs})
 matched=(optimal['status']=='found' and oracle is not None and optimal['cost']==oracle['cost']) or(optimal['status']=='no_roadmap_route' and oracle is None)
 if optimal['status']=='found':
  qs=[g['nodes'][i] for i in optimal['path']];optimal.update(states=qs,removed_names=scene.names(optimal['mask']),continuous_check=scene.verify_path(qs,optimal['mask']))
  if not optimal['continuous_check']:raise AssertionError('selected path failed independent conservative check')
 if optimal['status']!='censored' and not matched:raise AssertionError('exhaustive oracle mismatch')
 return {'joint':optimal,'oracle':oracle,'oracle_rows':oracle_rows,'oracle_match':matched,'greedy':greedy,'manual':manual,'intact':path_search(g,adj,allowed=0)}

 write(out/(base+('-replay' if replay else '')+'-result.json'),result);print(json.dumps({'case':name,'seed':seed,'nodes':len(g['nodes']),'edges':len(g['edges']),'joint':result['comparison']['joint'].get('cost'),'oracle_match':result['comparison']['oracle_match']}),flush=True)
