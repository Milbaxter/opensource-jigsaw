"""Download only exact public URLs pinned in native-source/manifest.json; no build/run."""
import hashlib,json,pathlib,urllib.request
R=pathlib.Path(__file__).resolve().parent
for row in json.loads((R/'native-source/manifest.json').read_text()):
 out=R/'native-source'/row['path'];out.parent.mkdir(parents=True,exist_ok=True)
 if out.exists() and hashlib.sha256(out.read_bytes()).hexdigest()==row['sha256']:continue
 data=urllib.request.urlopen(row['url'],timeout=30).read()
 if len(data)!=row['bytes'] or hashlib.sha256(data).hexdigest()!=row['sha256']:raise RuntimeError('pinned download mismatch: '+row['path'])
 out.write_bytes(data)
inputs=json.loads((R/'kernel-inputs.json').read_text());out=R/'inputs/test_adaptive.fcstd';out.parent.mkdir(exist_ok=True)
if not out.exists() or hashlib.sha256(out.read_bytes()).hexdigest()!=inputs['fixture_sha256']:
 data=urllib.request.urlopen(inputs['source_url'],timeout=30).read()
 if hashlib.sha256(data).hexdigest()!=inputs['fixture_sha256']:raise RuntimeError('fixture hash mismatch')
 out.write_bytes(data)
print('Exact native sources and CC BY fixture verified/downloaded; no execution.')
