"""Read-only primitive extraction. Does not load FreeCAD or generate toolpaths."""
import argparse,hashlib,json,pathlib,xml.etree.ElementTree as E,zipfile
ROOT=pathlib.Path(__file__).resolve().parent
parser=argparse.ArgumentParser();parser.add_argument('--fixture',type=pathlib.Path,default=ROOT/'inputs/test_adaptive.fcstd')
FIXTURE=parser.parse_args().fixture
EXPECTED='34207b128f433c53b68f09e58a14412eac6eff36bb6f25c7baaff659bf5d7375'
raw=FIXTURE.read_bytes();assert hashlib.sha256(raw).hexdigest()==EXPECTED
with zipfile.ZipFile(FIXTURE) as z: xml=z.read('Document.xml')
root=E.fromstring(xml)
objs={o.attrib['name']:o for o in root.findall('./ObjectData/Object')}
def p(o,n):return objs[o].find('./Properties/Property[@name="'+n+'"]')
def number(o,n):return float(p(o,n).find('Float').attrib['value'])
boxes={}
for name in ['Box','Box001','Box002']:
 pos=p(name,'Placement').find('PropertyPlacement').attrib
 assert [float(pos['Q'+str(i)]) for i in range(4)]==[0,0,0,1]
 boxes[name]={k:number(name,k) for k in ['Length','Width','Height']}
 boxes[name].update({k:float(pos[k]) for k in ['Px','Py','Pz']})
assert p('Cut','Base').find('Link').attrib['value']=='Box001'
assert p('Cut','Tool').find('Link').attrib['value']=='Box002'
assert [x.attrib['value'] for x in p('Fusion','Shapes').findall('LinkList/Link')]==['Box','Cut']
for name in ['Cut','Fusion']:
 pos=p(name,'Placement').find('PropertyPlacement').attrib
 assert [float(pos[k]) for k in ['Px','Py','Pz','Q0','Q1','Q2','Q3']]==[0,0,0,0,0,0,1]
a,b,h=[boxes[n] for n in ['Box','Box001','Box002']]
assert a['Px']+a['Length']==b['Px'] and b['Py']<a['Py']<a['Py']+a['Width']<b['Py']+b['Width']
assert b['Px']<h['Px']<h['Px']+h['Length']<b['Px']+b['Length']
assert b['Py']<h['Py']<h['Py']+h['Width']<b['Py']+b['Width']
def rect(v):
 x,y=v['Px'],v['Py'];X,Y=x+v['Length'],y+v['Width'];return [[x,y],[X,y],[X,Y],[x,Y]]
x=a['Px']; y=a['Py'];Y=y+a['Width']; xj=b['Px']; xl=xj+b['Length']; yl=b['Py'];yu=yl+b['Width']
combined=[[x,y],[xj,y],[xj,yl],[xl,yl],[xl,yu],[xj,yu],[xj,Y],[x,Y]]
shapes={'upper_face':[rect(a)],'lower_face_with_hole':[rect(b),list(reversed(rect(h)))],'projected_union_with_hole':[combined,list(reversed(rect(h)))]}
settings=dict(toolDiameter=5.,tolerance=.1,stockToLeave=0.,forceInsideOut=False,finishingProfile=False,keepToolDownDistRatio=3.,opType=0,helixRampTargetDiameter=0.,helixRampMinDiameter=0.)
workloads=[]
for name,paths in shapes.items():
 for stepover in [.2,.75]:
  workloads.append(dict(id=f'{name}_s{int(stepover*100)}',paths=paths,stockPaths=[[[0.,0.],[40.,0.],[40.,25.],[0.,25.]]],clearedPaths=[],settings={**settings,'stepOverFactor':stepover}))
record=dict(scope='Actual upstream fixture Boolean footprints as explicit planar native-kernel inputs; NOT full FreeCAD/OCC job replay or customer production program.',source_url='https://raw.githubusercontent.com/FreeCAD/FreeCAD/691a041981b7972d3170ceadb96163019712d2b0/src/Mod/CAM/CAMTests/test_adaptive.fcstd',fixture_sha256=EXPECTED,document_xml_sha256=hashlib.sha256(xml).hexdigest(),license='CC-BY-4.0',attribution='FreeCAD contributors; file LastModifiedBy Russell Johnson',units='millimetres',primitive_metadata=boxes,workloads=workloads)
(ROOT/'kernel-inputs.json').write_text(json.dumps(record,indent=2,allow_nan=False)+'\n')
print('Extracted six fixed planar input configurations; no planner called.')
