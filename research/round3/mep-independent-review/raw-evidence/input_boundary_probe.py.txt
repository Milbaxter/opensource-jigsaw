root=pathlib.Path(sys.argv[1]).resolve() if len(sys.argv)>1 else pathlib.Path(__file__).resolve().parent.parent/'round3-c'
spec=importlib.util.spec_from_file_location('frozen_mep',root/'prototype.py');p=importlib.util.module_from_spec(spec);spec.loader.exec_module(p)
records=[]

 records.append(item)
out={'source_sha256':hashlib.sha256((root/'prototype.py').read_bytes()).hexdigest(),'probe_sha256':hashlib.sha256(pathlib.Path(__file__).read_bytes()).hexdigest(),'original_files_unchanged':True,'checks':records,'scope':'Additional adversarial tests after freeze; not among original generated fixtures and not a change to measured runs.'}
